cmake_minimum_required(VERSION 3.16)
project(DinoRunner LANGUAGES CXX)
add_executable(dino WIN32 src/main.cpp)
target_compile_features(dino PRIVATE cxx_std_17)
target_compile_definitions(dino PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(dino PRIVATE user32 gdi32)
if(MSVC)
  target_compile_options(dino PRIVATE /W4 /permissive- /EHsc)
endif()
